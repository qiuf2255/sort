cmake_minimum_required(VERSION 3.20)
project(modern_sorting LANGUAGES CXX)
add_executable(sort_demo main.cpp)
set_target_properties(sort_demo PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(sort_demo PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(sort_demo PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME sorting_correctness COMMAND sort_demo)
